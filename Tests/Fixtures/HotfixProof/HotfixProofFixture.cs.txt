#if HOTFIX_PROOF_V1 || HOTFIX_PROOF_V2
using System.Threading;
using System.Threading.Tasks;
using Client.Contracts;
using Client.Game.Protocol;

namespace Client.Game.Proof
{
    public sealed class Entry : IHotfixEntry
    {
        public Task InitializeAsync(IClientContext context, CancellationToken token)
        {
            token.ThrowIfCancellationRequested();
            var registry = new GameMessageRegistry();
            if (registry.TryGetParser(101, out _) || registry.TryGetResponseId(101, out _))
                throw new System.InvalidOperationException("The formal business admission must remain empty.");
            context.Log.Write(ClientLogLevel.Info, "PROTOCOL_PROOF", "empty-registry");
#if HOTFIX_PROOF_V1
            context.Log.Write(ClientLogLevel.Info, "HOTFIX_PROOF", "proof-v1");
#elif HOTFIX_PROOF_V2
            context.Log.Write(ClientLogLevel.Info, "HOTFIX_PROOF", "proof-v2:" + new AddedAnswer().Compute());
#endif
            context.ReportReady();
            return Task.CompletedTask;
        }

        public Task ShutdownAsync(CancellationToken token) => Task.CompletedTask;
    }

#if HOTFIX_PROOF_V2
    public sealed class AddedAnswer
    {
        public int Compute()
        {
            int value = 6;
            return value == 6 ? value * 7 : -1;
        }
    }
#endif
}
#endif
