using System;
using System.Collections.Generic;
using System.Threading;
using System.Threading.Tasks;
using Client.Contracts;
using Client.Game.Protocol.F6Probe;
using Google.Protobuf;


namespace Client.Game.F6Probe
{
    public struct HotStruct { public int Value; }
    public sealed class Entry : IHotfixEntry
    {
        public async Task InitializeAsync(IClientContext context, CancellationToken token)
        {
            var values = new List<HotStruct> { new HotStruct { Value = 42 } };
            var messages = new Dictionary<uint, IMessage> { [101] = ProtocolProbe.CreateMessage() };
            await Task.Yield();
            token.ThrowIfCancellationRequested();
            if (values[0].Value != 42 || messages[101].CalculateSize() == 0) throw new InvalidOperationException("Hot struct/Dictionary IMessage generic failed.");
            ProtocolProbe.Run();
            context.Log.Write(ClientLogLevel.Info, "F6_AOT_PROBE", "list-hot-struct,dictionary-imessage,async-task,hot-repeated,hot-map,registered-hot-parser,unknown-fields,empty-registry");
            context.ReportReady();
        }
        public Task ShutdownAsync(CancellationToken token) => Task.CompletedTask;
    }
}
